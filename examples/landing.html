Quickstart

Add the CDN and a minimal HTML block.

CDN

<link rel="stylesheet" href="https://unpkg.com/@zumer/orbit@latest/dist/orbit.css">
<script src="https://unpkg.com/@zumer/orbit@latest/dist/orbit.js"></script>

HTML

<div class="bigbang">
  <div class="gravity-spot">
    <div class="orbit-4">
      <o-progress value="72"></o-progress>
    </div>
  </div>
</div>

Integrations

Works with React, Vue, Svelte. Export to image with SnapDOM.

// React
import '@zumer/orbit/style';
import '@zumer/orbit';

function Gauge({ value }) {
  return (
    <div className="bigbang">
      <div className="gravity-spot">
        <div className="orbit-4">
          <o-progress value={value}></o-progress>
        </div>
      </div>
    </div>
  );
}
